        if (raw) {

          if (parsed.teachers) db.teachers = parsed.teachers;

    async function syncFromSupabase(classCode) {
      if (!supabaseClient) return;
      try {
        const [{ data: tData }, { data: cData }, { data: sData }, { data: aData }] = await Promise.all([
          supabaseClient.from('teachers').select('*'),
          supabaseClient.from('classes').select('*'),
          classCode ? supabaseClient.from('students').select('*').eq('class_code', classCode) : supabaseClient.from('students').select('*'),
          classCode ? supabaseClient.from('assignments').select('*').eq('class_code', classCode) : supabaseClient.from('assignments').select('*')
        ]);
        if (tData && tData.length) db.teachers = tData;
        if (cData && cData.length) db.classes = cData;
        if (sData) {
          db.students = db.students.filter(s => s.class_code !== classCode).concat(sData);
        }
        if (aData) {
          db.assignments = db.assignments.filter(a => a.class_code !== classCode).concat(aData);
        }
        saveLocalDB();
      } catch (err) {
        console.warn("Supabase fetch skipped:", err);
      }
    }

    async function verifyOnlineDictionary(word, unit) {
      if (!navigator.onLine) return { status: "offline" };
      if (unit.examples.includes(word) || unit.nonExamples.includes(word)) return { status: "valid" };
      if (dictionaryCache.has(word)) return { status: dictionaryCache.get(word) ? "valid" : "invalid" };

      try {
        if (CONFIG.MERRIAM_WEBSTER_KEY) {
          const mwRes = await fetch(`https://www.dictionaryapi.com/api/v3/references/sd2/json/${encodeURIComponent(word)}?key=${CONFIG.MERRIAM_WEBSTER_KEY}`);
          if (!mwRes.ok) return { status: "offline" };
          const mwData = await mwRes.json();
          const exactEntry = Array.isArray(mwData) && mwData.some(item => item && typeof item === "object" && item.meta);
          dictionaryCache.set(word, exactEntry);
          return { status: exactEntry ? "valid" : "invalid" };
        }

        const res = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`);
        if (res.status === 404) {
          dictionaryCache.set(word, false);
          return { status: "invalid" };
        }
        if (!res.ok) return { status: "offline" };

        const data = await res.json();
        const isExactHeadword = Array.isArray(data) && data.some(entry =>
          entry &&
          typeof entry.word === "string" &&
          entry.word.toLowerCase() === word &&
          Array.isArray(entry.meanings) &&
          entry.meanings.length > 0
        );

        dictionaryCache.set(word, isExactHeadword);
        return { status: isExactHeadword ? "valid" : "invalid" };
      } catch (err) {
        return { status: "offline" };
      }
    }

    function getRandomItem(array) { return array[Math.floor(Math.random() * array.length)]; }

    function currentUnitMorphemes() {
      const unit = CURRICULUM[currentSetKey][currentUnitIndex];
      const cleaned = unit.id.replace(/\(.*?\)/g, '');
      return cleaned
        .split(/[/,]/)
        .map(part => part.replace(/[^a-zA-Z]/g, '').toLowerCase())
        .filter(Boolean)
        .sort((a, b) => b.length - a.length);
    }

    function addTag(word, type) {
      const container = type === 'example' ? document.getElementById('examplesContainer') : document.getElementById('nonExamplesContainer');
      const existing = Array.from(container.children).some(el => el.textContent.toLowerCase().includes(word.toLowerCase()));
      if (existing) return;
      const tag = document.createElement('span');
      tag.className = `tag tag-${type}`;
      tag.textContent = (type === 'example' ? '✓ ' : '⚠️ ') + word;

    }

    async function submitWord() {
      if (!navigator.onLine) {
        updateOnlineBanner();
        speakOwly(OWLY_DIALOGUE.offline());
        return;
      }

      const input = document.getElementById('wordInput');
      const btn = document.getElementById('checkWordBtn');
      const word = input.value.trim().toLowerCase().replace(/[^a-z]/g, '');
      if (!word) return;

      const unit = CURRICULUM[currentSetKey][currentUnitIndex];
      const morphemeParts = currentUnitMorphemes();

      if (unit.nonExamples.includes(word)) {
        addTag(word, 'nonexample');
        speakOwly(getRandomItem(OWLY_DIALOGUE.falseRoot)(word, unit.id));
        input.value = '';
        return;
      }

      btn.disabled = true;
      btn.textContent = "Checking...";
      const check = await verifyOnlineDictionary(word, unit);
      btn.disabled = false;
      btn.textContent = "Check Word";

      if (check.status === "offline") {
        speakOwly(OWLY_DIALOGUE.offline());
        return;
      }

      if (check.status !== "valid") {
        speakOwly(getRandomItem(OWLY_DIALOGUE.notInDictionary)(word));
        input.value = '';
        return;
      }

      const isKnownExample = unit.examples.includes(word);
      const matchesKey = morphemeParts.some(k => {
        if (unit.type === "Prefix") return word.startsWith(k) && word.length >= k.length + 2;
        if (unit.type === "Suffix") return word.endsWith(k) && word.length >= k.length + 2;
        return word.includes(k) && word.length >= k.length + 1;
      });

      if (isKnownExample || matchesKey) {
        addTag(word, 'example');
        const uKey = getUnitKey();
        if (!activeStudent.words_by_unit[uKey]) activeStudent.words_by_unit[uKey] = [];

        if (!activeStudent.words_by_unit[uKey].includes(word)) {
          activeStudent.words_by_unit[uKey].push(word);
          await persistActiveStudent();
          renderMorphemeSelector();
          updateUnitProgressUI(true, word);
        } else {
          speakOwly(`"${word}" is already saved for ${unit.id}! Try finding a new word.`);
        }
      } else {
        speakOwly(getRandomItem(OWLY_DIALOGUE.hints)(word));
      }

      input.value = '';
    }

    function launchGradedTest(scope, questionCount, title, assignmentId) {
      let pool = [];
      if (scope === 'set1' || scope === 'set2' || scope === 'set3') {
        pool = CURRICULUM[scope];
      } else {
        const [sKey, uId] = scope.split('_');
        const found = (CURRICULUM[sKey] || []).find(u => u.id === uId);
        pool = found ? [found] : CURRICULUM.set1;
      }

      const questions = [];
      for (let i = 0; i < questionCount; i++) {
        const unit = pool[i % pool.length];
        const qType = i % 3;

        if (qType === 0) {
          // Question Type 1: Morpheme Meaning
          const distractors = CURRICULUM.set1
            .filter(u => u.meaning !== unit.meaning)
            .sort(() => Math.random() - 0.5)
            .slice(0, 2)
            .map(u => u.meaning);
          questions.push({
            prompt: `What is the meaning of the ${unit.type.toLowerCase()} "${unit.id}"?`,
            correct: unit.meaning,
            options: [unit.meaning, ...distractors].sort(() => Math.random() - 0.5)
          });
        } else if (qType === 1) {
          // Question Type 2: Identify True Affixed Word vs False Root
          const correct = getRandomItem(unit.examples);
          const wrong1 = getRandomItem(unit.nonExamples) || "uncle";
          const wrong2 = "mountain";
          questions.push({
            prompt: `Which word truly uses "${unit.id}" (${unit.meaning})?`,
            correct,
            options: [correct, wrong1, wrong2].sort(() => Math.random() - 0.5)
          });
        } else {
          // Question Type 3: Spot the False Root ("Uncle Rule")
          const falseRoot = getRandomItem(unit.nonExamples) || "uncle";
          const trueWords = unit.examples.slice(0, 2);
          questions.push({
            prompt: `Spot the trickster! Which word does NOT use "${unit.id}" to mean "${unit.meaning}"?`,
            correct: falseRoot,
            options: [falseRoot, ...trueWords].sort(() => Math.random() - 0.5)
          });
        }
      }

      activeQuiz = {
        questions,
        currentIndex: 0,
        score: 0,
        assignmentId,
        title
      };

      document.getElementById('lessonWorkspace').style.display = 'none';
      document.getElementById('quizWorkspace').style.display = 'block';
      renderQuizQuestion();
      speakOwly(`Starting ${title}! Read each question carefully.`);
    }

    function renderQuizQuestion() {
      const q = activeQuiz.questions[activeQuiz.currentIndex];
      document.getElementById('quizTitle').textContent = activeQuiz.title;
      document.getElementById('quizStepCounter').textContent = `Question ${activeQuiz.currentIndex + 1} of ${activeQuiz.questions.length}`;
      document.getElementById('quizQuestion').textContent = q.prompt;
      document.getElementById('quizNextBtn').style.display = 'none';

      const optContainer = document.getElementById('quizOptions');
      optContainer.innerHTML = '';

      q.options.forEach(opt => {
        const btn = document.createElement('button');
        btn.className = 'quiz-opt-btn';
        btn.textContent = opt;
        btn.onclick = () => handleQuizAnswer(btn, opt, q.correct);
        optContainer.appendChild(btn);
      });
    }

    function handleQuizAnswer(clickedBtn, chosen, correct) {
      const allBtns = document.querySelectorAll('#quizOptions .quiz-opt-btn');
      allBtns.forEach(b => {
        b.disabled = true;
        if (b.textContent === correct) {
          b.style.background = '#81B29A';
          b.style.color = 'white';
        }
      });

      if (chosen === correct) {
        activeQuiz.score++;
        speakOwly("Hoot! Spot on!");
      } else {
        clickedBtn.style.background = '#E07A5F';
        clickedBtn.style.color = 'white';
        speakOwly(`Not quite—the correct answer is "${correct}".`);
      }

      const nextBtn = document.getElementById('quizNextBtn');
      nextBtn.style.display = 'inline-block';
      nextBtn.textContent = activeQuiz.currentIndex + 1 < activeQuiz.questions.length ? "Next Question →" : "Finish & Save Score";
    }

    async function nextQuizQuestion() {
      activeQuiz.currentIndex++;
      if (activeQuiz.currentIndex < activeQuiz.questions.length) {
        renderQuizQuestion();
        return;
      }

      const pct = Math.round((activeQuiz.score / activeQuiz.questions.length) * 100);
      const resultEntry = {
        assignment_id: activeQuiz.assignmentId,
        title: activeQuiz.title,
        score: pct,
        raw: `${activeQuiz.score}/${activeQuiz.questions.length}`,
        completed_at: new Date().toLocaleDateString()
      };

      if (activeStudent) {
        if (!activeStudent.test_results) activeStudent.test_results = [];
        if (activeQuiz.assignmentId) {
          activeStudent.test_results = activeStudent.test_results.filter(r => r.assignment_id !== activeQuiz.assignmentId);
        }
        activeStudent.test_results.push(resultEntry);
        await persistActiveStudent();
        renderAssignedTasks();
      }

      document.getElementById('quizStepCounter').textContent = "Test Complete!";
      document.getElementById('quizQuestion').textContent = `You scored ${activeQuiz.score} out of ${activeQuiz.questions.length} (${pct}%)!`;
      document.getElementById('quizOptions').innerHTML = '';
      document.getElementById('quizNextBtn').style.display = 'none';
      speakOwly(`Awesome work! You scored ${pct} percent on ${activeQuiz.title}!`);
    }

      const raw = document.getElementById('bulkStudentNames').value;
      const lines = raw.split('\n').map(l => l.trim()).filter(Boolean);
      if (!lines.length || !activeClassCode) return;

      const newStudents = lines.map((name, idx) => ({
        id: `stu_${Date.now()}_${idx}`,
        class_code: activeClassCode,
        name,
        avatar: ANIMAL_CHARACTERS[idx % ANIMAL_CHARACTERS.length].emoji,
        pin: [],
        words_by_unit: {},
        test_results: []
      }));

      db.students.push(...newStudents);
      saveLocalDB();
      if (supabaseClient) await supabaseClient.from('students').upsert(newStudents);

      document.getElementById('bulkStudentNames').value = '';
      renderTeacherRosterTable();
    }

    async function createAssignment() {
      const title = document.getElementById('assignTitle').value.trim();
      const scope = document.getElementById('assignScope').value;
      const question_count = parseInt(document.getElementById('assignQuestionCount').value, 10);
      const student_id = document.getElementById('assignTargetStudent').value || null;
      const due_date = document.getElementById('assignDueDate').value;
      if (!title || !activeClassCode) return;

      const asg = {
        id: `asg_${Date.now()}`,
        class_code: activeClassCode,
        student_id,
        title,
        scope,
        question_count,
        due_date
      };

      db.assignments.push(asg);
      saveLocalDB();
      if (supabaseClient) await supabaseClient.from('assignments').upsert(asg);

      document.getElementById('assignTitle').value = '';
      renderTeacherAssignmentsTable();
    }

    function renderTeacherAssignmentsTable() {
      const tbody = document.getElementById('teacherAssignmentsTableBody');
      tbody.innerHTML = '';
      const list = db.assignments.filter(a => a.class_code === activeClassCode);
      if (!list.length) {
        tbody.innerHTML = `<tr><td colspan="4" style="color:var(--color-text-muted);">No assignments yet.</td></tr>`;
        return;
      }
      list.forEach(a => {
        const stu = a.student_id ? db.students.find(s => s.id === a.student_id) : null;
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td><strong>${a.title}</strong></td>
          <td>${stu ? `${stu.avatar} ${stu.name}` : 'Entire Class'}</td>
          <td>${a.due_date || '—'}</td>
          <td><button class="btn-outline" onclick="deleteAssignment('${a.id}')">Remove</button></td>
        `;
        tbody.appendChild(tr);
      });
    }

    async function deleteAssignment(id) {
      db.assignments = db.assignments.filter(a => a.id !== id);
      saveLocalDB();
      if (supabaseClient) await supabaseClient.from('assignments').delete().eq('id', id);
      renderTeacherAssignmentsTable();
    }

    function renderStaffTable() {
      const tbody = document.getElementById('teacherStaffTableBody');
      tbody.innerHTML = '';
      db.teachers.forEach(t => {
        const tr = document.createElement('tr');
        tr.innerHTML = `<td><strong>${t.name}</strong></td><td>${t.email}</td><td>${t.is_admin ? 'Admin' : 'Teacher'}</td>`;
        tbody.appendChild(tr);
      });
    }

    async function authorizeTeacher() {
      const email = document.getElementById('newTeacherEmailInput').value.trim().toLowerCase();
      const name = document.getElementById('newTeacherNameInput').value.trim() || "Classroom Teacher";
      if (!email) return;

      const teacher = { email, name, is_admin: false };
      if (!db.teachers.some(t => t.email === email)) {
        db.teachers.push(teacher);
        saveLocalDB();
        if (supabaseClient) await supabaseClient.from('teachers').upsert(teacher);
      }
      document.getElementById('newTeacherEmailInput').value = '';
      document.getElementById('newTeacherNameInput').value = '';
      renderStaffTable();
    }

    function exportClassCSV() {
      const roster = db.students.filter(s => s.class_code === activeClassCode);
      const rows = [["Class Code", "Student Name", "Avatar", "Mastered Units", "Total Words", "Average Test Score"]];
      roster.forEach(s => {
        const results = s.test_results || [];
        const avg = results.length ? `${Math.round(results.reduce((acc, r) => acc + r.score, 0) / results.length)}%` : "N/A";
        rows.push([activeClassCode, s.name, s.avatar, getMasteredUnitsCount(s), getTotalStudentWords(s), avg]);
      });
      downloadCSV(rows, `${activeClassCode}_class_gradebook.csv`);
    }

    function downloadCSV(rows, filename) {
      const csvContent = "data:text/csv;charset=utf-8," + rows.map(e => e.join(",")).join("\n");
      const link = document.createElement("a");
      link.setAttribute("href", encodeURI(csvContent));
      link.setAttribute("download", filename);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }
